HỦY LISTENER CŨ // -------------------------------------------------------- if (appDataUnsub) { try { appDataUnsub(); } catch (e) {} appDataUnsub = null; } isAppDataLoadedFromCloud = false; // -------------------------------------------------------- // 2. /* * KHÔNG ghi đè appData bằng dữ liệu lỗi. */ return; } const cloudJSON = JSON.stringify(cloudData); const localJSON = JSON.stringify(appData); // ---------------------------------------------------- // Firebase giống local -> không làm gì // ---------------------------------------------------- if (cloudJSON === localJSON) { lastCloudAppDataJSON = cloudJSON; isAppDataLoadedFromCloud = true; console.log( "[SYNC] Local và Cloud đã đồng bộ." ); return; } // ==================================================== // SNAPSHOT NÀY CÓ THỂ LÀ KẾT QUẢ CỦA CHÍNH SAVE // ==================================================== if (cloudJSON === lastSavedAppDataJSON) { console.log( "[SYNC] Firebase xác nhận bản save mới nhất." ); lastCloudAppDataJSON = cloudJSON; isAppDataLoadedFromCloud = true; return; } // ==================================================== // CLOUD KHÁC LOCAL // ==================================================== /* * Chỉ lấy Cloud làm nguồn mới nếu: * * - Chưa có thao tác save đang chờ * - Không đang upload * * Nếu đang save, KHÔNG được để snapshot cũ * ghi đè dữ liệu người dùng vừa sửa. */ if (isAppDataSaving || pendingAppDataSave) { console.warn( "[SYNC] Bỏ qua snapshot vì local đang có thay đổi chưa upload." ); lastCloudAppDataJSON = cloudJSON; saveAppDataToLocal(); isAppDataLoadedFromCloud = true; renderHome(); }, (err) => { console.error( "[SYNC] Firebase realtime error:", err ); /* * Firebase lỗi KHÔNG được reset appData. * * Dữ liệu hiện tại vẫn giữ nguyên. */ } ); } catch (e) { console.error( "[SYNC] Không thể khởi tạo Firebase listener:", e ); /* * KHÔNG reset: * * appData = { subjects: [] }; * * vì sẽ gây mất dữ liệu. */ } } // ============================================================ // FORCE SAVE // ============================================================ async function forceSaveAppDataToCloud() { if ( !currentUser || currentUser.role !== "admin" ) { console.warn( "[SYNC] Không có quyền admin để lưu appData." ); /* * Dùng setDoc với merge:false có chủ đích: * * Document my_data chính là "database đề". * * Nhưng quan trọng là snapshot cũ không còn được phép * ghi ngược vào local trong lúc save. */ await window.firebaseSetDoc( docRef, dataToSave ); } finally { isAppDataSaving = false; } } // ============================================================ // SAVE APP DATA // ============================================================ function saveAppDataToCloud() { if ( !currentUser || currentUser.role !== "admin" ) { return; } // -------------------------------------------------------- // 1. HỦY TIMER CŨ // -------------------------------------------------------- if (syncTimeout) { clearTimeout(syncTimeout); syncTimeout = null; } // -------------------------------------------------------- // 4. /* * Firebase lỗi -> giữ local. */ if (localPersonalData) { userPersonalData = localPersonalData; } else { userPersonalData = { mistakes: [], habits: {} }; } } updateMistakeBadge(); renderHabitTracker(); } // ============================================================ // SAVE PERSONAL DATA // ============================================================ async function saveUserPersonalData(silent = false) { if (!currentUser || !currentUser.username) { return false; } const userKey = currentUser.username; const localKey = "vmu_personal_" + userKey; // -------------------------------------------------------- // LOCAL FIRST // -------------------------------------------------------- try { localStorage.setItem( localKey, JSON.stringify(userPersonalData) );
