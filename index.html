import { db } from './firebase-config'; import { collection, doc, updateDoc, deleteDoc, onSnapshot } from 'firebase/firestore'; /** * 1. ĐỒNG BỘ DỮ LIỆU THỜI GIAN THỰC (TỐC ĐỘ CAO) * - Đã xóa hoàn toàn dữ liệu mẫu/mặc định (sinh lý máu). * - Sử dụng onSnapshot kết hợp Cache của Firestore giúp tải tức thì. * - Lấy đầy đủ tài khoản Admin và User thường. */ export function subscribeToUsers(onDataUpdate) { const usersRef = collection(db, 'users_list'); // Lắng nghe thay đổi thời gian thực - Tốc độ cực nhanh nhờ Cache Firestore const unsubscribe = onSnapshot(usersRef, (snapshot) => { const allUsers = snapshot.docs.map((docSnap) => ({ id: docSnap.id, ...docSnap.data() })); // Phân loại dữ liệu trực tiếp trên RAM (không gọi query DB nhiều lần) const admins = allUsers.filter(u => u.role === 'admin'); const pendingUsers = allUsers.filter(u => u.status === 'pending'); // Trả dữ liệu về giao diện ngay lập tức if (typeof onDataUpdate === 'function') { onDataUpdate({ allUsers, // Danh sách tất cả người dùng (gồm cả Admin) admins, // Danh sách Admin pendingUsers // Danh sách đang chờ duyệt }); } }, (error) => { console.error("Lỗi đồng bộ dữ liệu Firestore:", error); }); return unsubscribe; } /** * 2. PHÊ DUYỆT TÀI KHOẢN CHỜ (STATUS -> APPROVED) */ export async function approveUser(userId) { try { const userRef = doc(db, 'users_list', userId); await updateDoc(userRef, { status: 'approved' }); console.log("Đã phê duyệt tài khoản:", userId); } catch (error) { console.error("Lỗi khi phê duyệt:", error); } } /** * 3. XÓA TÀI KHOẢN (NẾU CẦN XÓA TÀI KHOẢN RÁC / MẪU) */ export async function deleteUser(userId) { try { const userRef = doc(db, 'users_list', userId); await deleteDoc(userRef); console.log("Đã xóa tài khoản thành công:", userId); } catch (error) { console.error("Lỗi khi xóa tài khoản:", error); } }
