async function handleLogin() { let user = document.getElementById('auth-username').value.trim(); let pass = document.getElementById('auth-password').value; let errEl = document.getElementById('auth-error-msg'); errEl.classList.add('hidden'); if (!user || !pass) { errEl.innerText = "Vui lòng nhập đầy đủ tên và mật khẩu!"; errEl.classList.remove('hidden'); return; } // Xử lý Admin đăng nhập bằng Mật khẩu cứng if (user.toLowerCase() === 'admin' || pass === ADMIN_PASSWORD) { if (pass === ADMIN_PASSWORD) { currentUser = { username: user, role: 'admin', status: 'approved' }; localStorage.setItem("vmu_current_user", JSON.stringify(currentUser)); document.getElementById('modal-auth').classList.add('hidden'); updateNavbarUser(); await loadAppData(); await loadUserPersonalData(); return; } } // Xử lý User thường kiểm tra trên Firebase Firestore try { if (window.firebaseDB) { let docRef = window.firebaseDoc(window.firebaseDB, "users_list", user); let snap = await window.firebaseGetDoc(docRef); if (snap.exists()) { let data = snap.data(); if (data.password === pass) { if (data.status !== 'approved') { errEl.innerText = "Tài khoản của bạn đang chờ Admin phê duyệt!"; errEl.classList.remove('hidden'); return; } currentUser = { username: user, role: 'member', status: 'approved' }; localStorage.setItem("vmu_current_user", JSON.stringify(currentUser)); document.getElementById('modal-auth').classList.add('hidden'); updateNavbarUser(); await loadAppData(); await loadUserPersonalData(); return; } else { errEl.innerText = "Mật khẩu không chính xác!"; errEl.classList.remove('hidden'); } } else { errEl.innerText = "Tài khoản không tồn tại!"; errEl.classList.remove('hidden'); } } else { errEl.innerText = "Không thể kết nối đến cơ sở dữ liệu Firebase!"; errEl.classList.remove('hidden'); } } catch (err) { console.error("Lỗi đăng nhập:", err); errEl.innerText = "Đã xảy ra lỗi kết nối: " + err.message; errEl.classList.remove('hidden'); } }
